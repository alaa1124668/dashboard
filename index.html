function parsePerformanceSheetData(rows) { const headers = rows[0].map(h => String(h || '').trim()); let riderIdIdx = headers.findIndex(c => c.toLowerCase().includes('rider id') || c.includes('كود')); let riderNameIdx = headers.findIndex(c => c.toLowerCase().includes('rider name') || c.includes('اسم')); let zoneIdx = headers.findIndex(c => c.toLowerCase().includes('zone') || c.includes('منطقة')); let supervisorIdx = headers.findIndex(c => c.includes('المشرف') || c.toLowerCase().includes('supervisor')); if (riderIdIdx === -1) riderIdIdx = 0; if (riderNameIdx === -1) riderNameIdx = 1; if (zoneIdx === -1) zoneIdx = 2; if (supervisorIdx === -1) supervisorIdx = 3; let parsedData = []; for (let r = 1; r < rows.length; r++) { const row = rows[r]; if (!Array.isArray(row)) continue; const riderId = row[riderIdIdx] !== undefined ? String(row[riderIdIdx]).trim() : ''; const riderName = row[riderNameIdx] !== undefined ? String(row[riderNameIdx]).trim() : ''; const zone = row[zoneIdx] !== undefined ? String(row[zoneIdx]).trim() : ''; const supervisor = row[supervisorIdx] !== undefined ? String(row[supervisorIdx]).trim() : ''; // تجاهل صف الهيدر الإضافي إن وجد if ((!riderId && !riderName) || riderId.toLowerCase() === 'rider id') continue; let workDays = 0, notWorkingDays = 0, absDays = 0, susDays = 0, totalHoursSec = 0; for (let i = 4; i < row.length; i++) { const cellVal = row[i]; if (cellVal === null || cellVal === undefined) continue; const val = String(cellVal).trim(); if (!val || val === 'NaN') continue; if (val.includes('غياب')) { absDays++; notWorkingDays++; } else if (val.includes('موقوف')) { susDays++; notWorkingDays++; } else if (val.includes('مش مختار')) { notWorkingDays++; } else { // تحويل آمن لقيم الوقت والساعات let parsedSec = 0; if (typeof cellVal === 'number' && !isNaN(cellVal)) { parsedSec = cellVal < 1 ? cellVal * 86400 : cellVal * 3600; } else if (val.includes(':')) { const parts = val.split(':').map(p => parseFloat(p) || 0); if (parts.length >= 2) { parsedSec = (parts[0] * 3600) + (parts[1] * 60) + (parts[2] || 0); } } else if (!isNaN(parseFloat(val))) { const num = parseFloat(val); parsedSec = num < 1 ? num * 86400 : num * 3600; } if (parsedSec > 0) { workDays++; totalHoursSec += parsedSec; } } } // تحويل الثواني المجمعة إلى ساعات ودقائق بضمان أرقام صحيحة const validSec = isNaN(totalHoursSec) ? 0 : totalHoursSec; const hours = Math.floor(validSec / 3600); const mins = Math.floor((validSec % 3600) / 60); parsedData.push({ "كود المندوب": riderId, "الاسم": riderName, "المنطقة": zone, "المشرف": supervisor, "أيام العمل": workDays, "أيام عدم العمل": notWorkingDays, "إجمالي الساعات": `${hours}:${mins < 10 ? '0' : ''}${mins}`, "ساعات بالثواني": validSec, "الغياب": absDays, "الإيقاف": susDays }); } processPerformanceSheetData(parsedData); }
